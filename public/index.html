Integration Snippets

lib/core/services/update_service.dart 
import 'dart:convert';
import 'package:http/http.dart' as http;

class UpdateService {
  static const String baseUrl = 'http://YOUR_VPS_IP:3000';
  static const String appId = 'toho_egs';

  /// Checks if a new version is available on the VPS
  static Future<Map<String, dynamic>?> checkForUpdate(String currentVersion) async {
    final uri = Uri.parse('$baseUrl/api/v1/releases/latest?app=$appId&platform=android&current_version=$currentVersion');
    
    try {
      final response = await http.get(uri);
      if (response.statusCode == 200) {
        final data = jsonDecode(response.body)['data'];
        
        if (data['has_update'] == true) {
          return {
            'hasUpdate': true,
            'isMandatory': data['is_mandatory'] ?? false,
            'latestVersion': data['latest_version'],
            'releaseTitle': data['release_title'],
            'releaseNotes': data['release_notes'],
            'downloadUrl': data['primary_asset']?['download_url'],
            'sha256': data['primary_asset']?['sha256'],
          };
        }
      }
    } catch (e) {
      print('Update check failed: $e');
    }
    return null;
  }
}
Terminal / Bash Script 
# 1. Check latest release metadata
curl -X GET "http://YOUR_VPS_IP:3000/api/v1/releases/latest?app=toho_egs&platform=android"

# 2. Download the latest APK directly
curl -L -O "http://YOUR_VPS_IP:3000/api/v1/releases/download/ASSET_ID/app.apk"

# 3. Upload a new file to universal storage
curl -X POST "http://YOUR_VPS_IP:3000/api/v1/storage/upload" \
  -H "x-api-key: YOUR_MASTER_KEY" \
  -F "bucket=firmware" \
  -F "file=@sensor_driver.bin"
Node.js / TypeScript Client 
import fetch from 'node-fetch';

async function checkRelease() {
  const res = await fetch('http://YOUR_VPS_IP:3000/api/v1/releases/latest?app=toho_egs');
  const json = await res.json();
  
  if (json.data.has_update) {
    console.log(`New version found: ${json.data.latest_version}`);
    console.log(`Download: ${json.data.primary_asset.download_url}`);
  } else {
    console.log('App is up to date.');
  }
}
